<script>

function tekshir(){

let jamiSavol = 15;

let fio = document.getElementById("fio").value.trim();
let sinf = document.getElementById("sinf").value;

if(fio === "" || sinf === ""){
    alert("Iltimos, o‘quvchining familiyasi va ismini hamda sinfini kiriting!");
    return;
}

let ball = 0;

let hisobot = "";

for(let i=1; i<=jamiSavol; i++){

let javob =
document.querySelector('input[name="q'+i+'"]:checked');

if(javob){

if(javob.value=="1"){

ball++;

hisobot += "✅ " + i +
"-savol : To'g'ri<br>";

}else{

hisobot += "❌ " + i +
"-savol : Noto'g'ri<br>";

}

}else{

hisobot += "⚠ " + i +
"-savol javob belgilanmagan.<br>";

}

}

let foiz = (ball/jamiSavol)*100;

let baho="";

if(foiz>=90){

baho="5";

}else if(foiz>=70){

baho="4";

}else if(foiz>=50){

baho="3";

}else{

baho="2";

}

document.getElementById("natija").innerHTML=

"<h2>TEST NATIJASI</h2>" +
"<b>O‘quvchi:</b> " + fio +
"<br><b>Sinf:</b> " + sinf +
"<br><br>" +
hisobot +

"<hr>" +

"<b>To'g'ri javoblar:</b> " + ball +

"<br><b>Jami savollar:</b> " + jamiSavol +

"<br><b>Foiz:</b> " + foiz.toFixed(1) + "%" +

"<br><br><b>Baho:</b> " + baho;


    window.testNatija = {
        fio: fio,
        sinf: sinf,
        ball: ball,
        jamiSavol: jamiSavol,
        foiz: foiz,
        baho: baho,
        sana: new Date().toLocaleString("uz-UZ")
    };
}


function excelgaSaqlash(){

    if(!window.testNatija){
        alert("Avval testni bajaring va «Natijani ko‘rish» tugmasini bosing!");
        return;
    }

    let n = window.testNatija;

    // Oldingi o'quvchilar natijalarini brauzer xotirasidan olamiz
    let natijalar = JSON.parse(localStorage.getItem("testNatijalari") || "[]");

    // Yangi natijani qo'shamiz
    natijalar.push({
        sana: n.sana,
        fio: n.fio,
        sinf: n.sinf,
        ball: n.ball,
        jamiSavol: n.jamiSavol,
        foiz: n.foiz.toFixed(1) + "%",
        baho: n.baho
    });

    // Brauzerda saqlab qo'yamiz
    localStorage.setItem("testNatijalari", JSON.stringify(natijalar));

    // Barcha o'quvchilar natijasini Excel jadvaliga tayyorlaymiz
    let jadval = [
        ["№", "Sana", "O‘quvchi F.I.O.", "Sinf",
         "To‘g‘ri javoblar", "Jami savollar", "Foiz", "Baho"]
    ];

    natijalar.forEach(function(r, index){
        jadval.push([
            index + 1,
            r.sana,
            r.fio,
            r.sinf,
            r.ball,
            r.jamiSavol,
            r.foiz,
            r.baho
        ]);
    });

    let worksheet = XLSX.utils.aoa_to_sheet(jadval);

    // Ustun kengliklari
    worksheet["!cols"] = [
        {wch:5},
        {wch:20},
        {wch:28},
        {wch:10},
        {wch:18},
        {wch:16},
        {wch:12},
        {wch:8}
    ];

    let workbook = XLSX.utils.book_new();
    XLSX.utils.book_append_sheet(workbook, worksheet, "Natijalar");

    XLSX.writeFile(workbook, "Barcha_oquvchilar_natijalari.xlsx");

    alert("Natija saqlandi!\nJami o‘quvchilar: " + natijalar.length);
}

// Barcha yig'ilgan natijalarni Excelga qayta chiqarish
function barchaNatijalarniExcelga(){

    let natijalar = JSON.parse(localStorage.getItem("testNatijalari") || "[]");

    if(natijalar.length === 0){
        alert("Hozircha saqlangan o‘quvchi natijalari yo‘q.");
        return;
    }

    let jadval = [
        ["№", "Sana", "O‘quvchi F.I.O.", "Sinf",
         "To‘g‘ri javoblar", "Jami savollar", "Foiz", "Baho"]
    ];

    natijalar.forEach(function(r, index){
        jadval.push([
            index + 1,
            r.sana,
            r.fio,
            r.sinf,
            r.ball,
            r.jamiSavol,
            r.foiz,
            r.baho
        ]);
    });

    let worksheet = XLSX.utils.aoa_to_sheet(jadval);

    worksheet["!cols"] = [
        {wch:5},
        {wch:20},
        {wch:28},
        {wch:10},
        {wch:18},
        {wch:16},
        {wch:12},
        {wch:8}
    ];

    let workbook = XLSX.utils.book_new();
    XLSX.utils.book_append_sheet(workbook, worksheet, "Natijalar");

    XLSX.writeFile(workbook, "Barcha_oquvchilar_natijalari.xlsx");
}
